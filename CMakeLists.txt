# STM32 Temperature Control System
cmake_minimum_required(VERSION 3.22)

# Set cross-compiler before project()
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
set(CMAKE_OBJCOPY arm-none-eabi-objcopy)
set(CMAKE_SIZE arm-none-eabi-size)

# Project settings
project(stm32_temp_control C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

# MCU settings
set(MCU_FAMILY STM32F1xx)
set(MCU_MODEL STM32F103xB)
set(CPU_PARAMETERS
    -mcpu=cortex-m3
    -mthumb
    -mfloat-abi=soft)

# Linker script
set(LINKER_SCRIPT ${CMAKE_SOURCE_DIR}/STM32F103C8TX_FLASH.ld)

# Sources - CubeMX generated HAL files
file(GLOB_RECURSE HAL_SOURCES
    "Drivers/STM32F1xx_HAL_Driver/Src/*.c")

# Application sources
set(PROJECT_SOURCES
    # Startup file
    Core/Startup/startup_stm32f103c8tx.s

    # HAL initialization (generated by CubeMX)
    Core/Src/main.c
    Core/Src/stm32f1xx_it.c
    Core/Src/stm32f1xx_hal_msp.c
    Core/Src/system_stm32f1xx.c
    Core/Src/syscalls.c
    Core/Src/sysmem.c
    Core/Src/gpio.c
    Core/Src/i2c.c
    Core/Src/tim.c
    Core/Src/usart.c

    # BSP layer
    Core/Application/bsp/bsp_pwm.c
    Core/Application/bsp/bsp_bluetooth.c

    # Driver layer
    Core/Application/driver/ds18b20.c
    Core/Application/driver/ssd1306.c

    # Control layer
    Core/Application/control/pid_controller.c
    Core/Application/control/temp_control.c
    Core/Application/control/system_task.c
)

# Include directories
set(PROJECT_INCLUDES
    Core/Inc
    Core/Application/bsp
    Core/Application/driver
    Core/Application/control
    Drivers/STM32F1xx_HAL_Driver/Inc
    Drivers/STM32F1xx_HAL_Driver/Inc/Legacy
    Drivers/CMSIS/Device/ST/STM32F1xx/Include
    Drivers/CMSIS/Include
)

# Executable
add_executable(${PROJECT_NAME} ${PROJECT_SOURCES} ${HAL_SOURCES})

# Include directories
target_include_directories(${PROJECT_NAME} PRIVATE ${PROJECT_INCLUDES})

# Compiler definitions
target_compile_definitions(${PROJECT_NAME} PRIVATE
    USE_HAL_DRIVER
    ${MCU_MODEL}
)

# Compiler options
target_compile_options(${PROJECT_NAME} PRIVATE
    ${CPU_PARAMETERS}
    -Wall
    -Wextra
    -Wpedantic
    -Wno-unused-parameter
    $<$<CONFIG:Debug>:-Os -g3 -ggdb>
    $<$<CONFIG:Release>:-Os -flto>
    -ffunction-sections
    -fdata-sections
)

# Linker options
target_link_options(${PROJECT_NAME} PRIVATE
    ${CPU_PARAMETERS}
    -T${LINKER_SCRIPT}
    -specs=nano.specs
    -lc
    -lm
    -Wl,-Map=${PROJECT_NAME}.map
    -Wl,--gc-sections
    -Wl,--print-memory-usage
)

# Print executable size
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${PROJECT_NAME}>
)

# Create hex file
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:${PROJECT_NAME}> ${PROJECT_NAME}.hex
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${PROJECT_NAME}> ${PROJECT_NAME}.bin
)
